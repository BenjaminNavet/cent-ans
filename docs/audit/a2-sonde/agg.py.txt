"""Aggregate a2probe JSON lines."""

import json
import statistics as st
import sys
from collections import Counter, defaultdict

path = sys.argv[1]
runs = [json.loads(line) for line in open(path)]
N = len(runs)
T = runs[0]["turns"]
MAJ = ["fac_france", "fac_england", "fac_burgundy", "fac_scotland", "fac_castile", "fac_aragon", "fac_portugal", "fac_flanders", "fac_brittany", "fac_navarre", "fac_hre", "fac_papacy"]
print(f"# {path}: {N} seeds x {T} turns, end {runs[0]['end']}, {st.mean(r['secs'] for r in runs):.1f}s/run")

# events per run
kinds = Counter()
for r in runs:
    kinds.update(r["kinds"])
print("\n## events (mean per campaign, per turn)")
for k, v in sorted(kinds.items(), key=lambda x: -x[1]):
    print(f"{k:22} {v / N:8.1f} {v / N / T:6.2f}")

# snapshot table for majors
snaps_keys = sorted({int(k) for r in runs for k in r["snaps"]})
facs = Counter()
for r in runs:
    for f in r["snaps"][str(T)]:
        facs[f] += 1
print("\n## factions alive at end (seeds):", dict(facs))
dead_all = Counter()
for r in runs:
    for f in r["snaps"]["0"]:
        if f not in r["snaps"][str(T)]:
            dead_all[f] += 1
print("## factions dead at end (count of seeds):", dict(dead_all))

fields = ["treasury", "income", "upkeep", "field_men", "garrison_men", "armies", "provinces", "settlements", "buildings", "techs", "unrest", "power", "wars", "agents"]
for f in ["fac_france", "fac_england", "fac_burgundy", "fac_scotland"]:
    print(f"\n## {f} (median over seeds; n alive)")
    print("turn  " + " ".join(f"{x[:9]:>9}" for x in fields))
    for t in snaps_keys:
        vals = [r["snaps"][str(t)].get(f) for r in runs if str(t) in r["snaps"]]
        vals = [v for v in vals if v]
        if not vals:
            print(f"{t:4} dead")
            continue
        print(f"{t:4}  " + " ".join(f"{st.median(v[x] for v in vals):9.0f}" for x in fields) + f"  n={len(vals)}")

# top faction by provinces per seed at end
print("\n## top-3 provinces at end per seed; share of top")
tot_prov = sum(v["provinces"] for v in runs[0]["snaps"]["0"].values())
for r in runs:
    end = r["snaps"][str(T)]
    top = sorted(end.items(), key=lambda x: -x[1]["provinces"])[:4]
    start = r["snaps"]["0"]
    print(r["seed"], [(f[4:], v["provinces"], start.get(f, {}).get("provinces", 0)) for f, v in top], f"alive={len(end)}")
print("start provinces top:", sorted(((f[4:], v["provinces"]) for f, v in runs[0]["snaps"]["0"].items()), key=lambda x: -x[1])[:10], "total", tot_prov, "factions", len(runs[0]["snaps"]["0"]))

# war
print("\n## wars")
print("new war pairs per campaign:", st.mean(r["new_wars"] for r in runs))
durs = [d[3] for r in runs for d in r["war_durations"]]
if durs:
    print("ended wars: n/campaign", len(durs) / N, "median duration", st.median(durs), "mean", round(st.mean(durs), 1))
ongoing = [T - w[2] for r in runs for w in r["ongoing_wars"]]
print("ongoing wars at end per campaign", len(ongoing) / N, "median age", st.median(ongoing) if ongoing else 0)
fe = []
for r in runs:
    turns = sum(d[3] for d in r["war_durations"] if {d[0], d[1]} == {"fac_england", "fac_france"})
    turns += sum(T - w[2] for w in r["ongoing_wars"] if {w[0], w[1]} == {"fac_england", "fac_france"})
    fe.append(turns / T * 100)
print("France-England war share %:", [round(x) for x in fe])
pairs = Counter()
for r in runs:
    for d in r["war_durations"]:
        pairs[(d[0][4:], d[1][4:])] += 1
    for w in r["ongoing_wars"]:
        pairs[(w[0][4:], w[1][4:])] += 1
print("most frequent war pairs:", pairs.most_common(12))

# battles
print("\n## battles")
bs = [b for r in runs for b in r["battles"]]
print("battles per campaign", len(bs) / N, "per turn", len(bs) / N / T)
att_win = sum(1 for b in bs if b["win"] == b["att"]) / max(1, len(bs))
print("attacker win rate", round(att_win * 100, 1))
bf = Counter()
bw = Counter()
for b in bs:
    for side in ("att", "def"):
        bf[b[side]] += 1
    bw[b["win"]] += 1
print("battles fought / won by faction:", [(f[4:], bf[f], bw[f], round(100 * bw[f] / bf[f])) for f, _ in bf.most_common(12)])
losses = [(b["la"] + b["lb"]) for b in bs]
print("mean total losses per battle", round(st.mean(losses)) if losses else 0)
ratio_one_sided = sum(1 for b in bs if min(b["la"], b["lb"]) == 0) / max(1, len(bs))
print("one-sided (a side lost 0)", round(ratio_one_sided * 100, 1), "%")
ptb = [sum(r["per_turn_battles"][i] for r in runs) / N for i in range(T)]
chunk = T // 10
print("battles per turn by decile:", [round(sum(ptb[i:i + chunk]) / chunk, 2) for i in range(0, T, chunk)])

print("\n## captures (province owner changes by new owner), mean per campaign")
cap = Counter()
for r in runs:
    cap.update(r["captures"])
print([(k[4:], round(v / N, 1)) for k, v in cap.most_common(12)])
print("owner changes per campaign", st.mean(r["owner_changes"] for r in runs))
rev = Counter()
for r in runs:
    rev.update(r["revolts"])
print("revolts by owner (mean):", [(k[4:], round(v / N, 1)) for k, v in rev.most_common(10)])
print("faction deaths:", Counter(fd[1] for r in runs for fd in r["faction_deaths"]).most_common(20))

# orders and recruitment
print("\n## AI orders (mean per campaign)")
od = Counter()
for r in runs:
    od.update(r["orders"])
for k, v in sorted(od.items()):
    print(f"  {k:45} {v / N:9.1f}")

print("\n## unit composition at end (mean count)")
cp = Counter()
for r in runs:
    cp.update(r["comps"][str(T)])
for k, v in cp.most_common():
    print(f"  {k:40} {v / N:8.1f}")
cp0 = Counter(runs[0]["comps"]["0"])
print("  start:", cp0.most_common())

print("\n## buildings completed (mean per campaign)")
bl = Counter()
for r in runs:
    bl.update(r["built"])
for k, v in bl.most_common():
    print(f"  {k:30} {v / N:8.1f}")
print("  never completed:", [b for b in json.load(open("buildings.json")) if b not in bl])

print("\n## technologies (share of alive factions owning at end)")
tc = Counter()
nf = 0
for r in runs:
    for f, ts in r["techs"].items():
        nf += 1
        tc.update(ts)
alltech = json.load(open("techs.json"))
for t in alltech:
    print(f"  {t:32} {100 * tc[t] / nf:5.0f} %")
maj_t = defaultdict(list)
for r in runs:
    for f in ["fac_france", "fac_england", "fac_burgundy"]:
        if f in r["techs"]:
            maj_t[f].append(len(r["techs"][f]))
print("  techs of majors at end:", {f: st.mean(v) for f, v in maj_t.items()}, "of", len(alltech))

print("\n## chronicle")
fired = Counter()
for r in runs:
    fired.update(r["fired"])
evts = json.load(open("events.json"))
hist = [e for e, k in evts.items() if k != "random"]
rnd = [e for e, k in evts.items() if k == "random"]
years = json.load(open("event_years.json"))
end_year = 1337 + T // 4
never = [(e, years.get(e)) for e in hist if e not in fired and years.get(e, 0) <= end_year]
hist = [e for e in hist if years.get(e, 0) <= end_year]
print(f"historical+chained events dated <= {end_year}: {len(hist)}, never fired in any seed: {len(never)}")
print("  ", never)
always = [e for e in hist if fired[e] == N]
print(f"  fired in every seed: {len(always)}")
titles = Counter()
for r in runs:
    titles.update(r["chronicle"])
print("  player (France) chronicle entries, top:", titles.most_common(15))
rtitles = json.load(open("event_titles.json"))
never_r = [e for e in rnd if not any(rtitles[e] in t for t in titles)]
print(f"random events: {len(rnd)}; never seen in France's journal: {len(never_r)}: {never_r}")

print("\n## objectives / scores")
for f in ["fac_france", "fac_england", "fac_burgundy"]:
    sc = [r["objectives"][f]["score"] for r in runs]
    dn = Counter(o for r in runs for o in r["objectives"][f]["done"])
    print(f, "score median", st.median(sc), "done:", dict(dn))
    print("   ex:", runs[0]["objectives"][f]["all"])
print("outcomes:", Counter(r["outcome"][:40] for r in runs))
print("\n## characters")
tr = Counter()
sk = Counter()
for r in runs:
    tr.update(r["traits"])
    sk.update(r["skills"])
print("alive chars", st.mean(r["chars_alive"] for r in runs))
print("traits (alive, mean):", [(k, round(v / N, 1)) for k, v in tr.most_common()])
print("skills learned (mean):", [(k, round(v / N, 1)) for k, v in sk.most_common()])
print("avg devastation", round(st.mean(r["avg_dev"] for r in runs), 1), "avg unrest", round(st.mean(r["avg_unrest"] for r in runs), 1), "heresy provinces", st.mean(r["heresy_provinces"] for r in runs))
print("france orders issued/refused", [r["france_orders"] for r in runs])
tax = Counter()
coin = Counter()
for r in runs:
    for f, v in r["snaps"][str(T)].items():
        tax[v["tax"]] += 1
        coin[v["coinage"]] += 1
print("tax rates at end:", dict(tax), "coinage:", dict(coin))
