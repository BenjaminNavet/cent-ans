//! A2 audit probe: AI vs AI campaigns, one JSON line per seed.
//! Usage: a2probe <turns> <seed...>
use std::cell::RefCell;
use std::collections::{BTreeMap, BTreeSet};
use std::path::Path;

use data_model::{FactionId, GameData};
use serde_json::{json, Value};
use sim_campaign::{CampaignState, EventKind, Order};

fn fid(s: &str) -> FactionId {
    FactionId::new(s).unwrap()
}

fn snapshot(state: &CampaignState, data: &GameData) -> Value {
    let mut out = serde_json::Map::new();
    for (id, f) in &state.factions {
        if !f.alive {
            continue;
        }
        let field_units: usize = state
            .armies
            .values()
            .filter(|a| &a.faction == id)
            .map(|a| a.units.len())
            .sum();
        let field_men: u32 = state
            .armies
            .values()
            .filter(|a| &a.faction == id)
            .map(|a| a.total_strength())
            .sum();
        let armies = state.armies.values().filter(|a| &a.faction == id).count();
        let garrison_men: u32 = state
            .settlements
            .values()
            .filter(|s| &s.controller == id)
            .map(|s| s.garrison_strength())
            .sum();
        let settlements = state
            .settlements
            .values()
            .filter(|s| &s.controller == id)
            .count();
        let buildings: usize = state
            .settlements
            .values()
            .filter(|s| &s.controller == id)
            .map(|s| s.buildings.len())
            .sum();
        let provinces = state.controlled_provinces(id);
        let owned = state
            .provinces
            .keys()
            .filter(|p| state.province_owner(p) == Some(id))
            .count();
        let unrest: f64 = if provinces.is_empty() {
            0.0
        } else {
            provinces
                .iter()
                .map(|p| f64::from(state.provinces[p].unrest))
                .sum::<f64>()
                / provinces.len() as f64
        };
        let _ = data;
        out.insert(
            id.as_str().to_owned(),
            json!({
                "treasury": f.treasury,
                "income": f.income_last_turn,
                "upkeep": f.upkeep_last_turn,
                "army_upkeep": f.army_upkeep_last_turn,
                "building_upkeep": f.building_upkeep_last_turn,
                "armies": armies,
                "field_units": field_units,
                "field_men": field_men,
                "garrison_men": garrison_men,
                "provinces": provinces.len(),
                "owned": owned,
                "settlements": settlements,
                "buildings": buildings,
                "techs": f.technologies.len(),
                "unrest": unrest,
                "tax": format!("{:?}", f.tax_rate),
                "coinage": format!("{:?}", f.coinage),
                "price": f.price_level,
                "wars": f.at_war_with.len(),
                "allies": f.allies.len(),
                "suzerain": f.suzerain.as_ref().map(|s| s.as_str().to_owned()),
                "papal": f.papal_favor,
                "power": state.faction_power(id),
                "agents": state.agents.agents.values().filter(|a| format!("{:?}", a).contains(id.as_str())).count(),
            }),
        );
    }
    Value::Object(out)
}

fn composition(state: &CampaignState) -> BTreeMap<String, u32> {
    let mut m = BTreeMap::new();
    for a in state.armies.values() {
        for u in &a.units {
            *m.entry(u.unit_type.as_str().to_owned()).or_default() += 1;
        }
    }
    for s in state.settlements.values() {
        for u in &s.garrison {
            *m.entry(format!("garrison:{}", u.unit_type.as_str()))
                .or_default() += 1;
        }
    }
    m
}

fn main() {
    let mut args = std::env::args().skip(1);
    let turns: u32 = args.next().and_then(|s| s.parse().ok()).unwrap_or(200);
    let seeds: Vec<u64> = args.filter_map(|s| s.parse().ok()).collect();
    let root = Path::new("/Users/jean_hubert/dev/game_project/data");
    let (data, _) = GameData::load(root).expect("data");
    for seed in seeds {
        let line = run(&data, seed, turns);
        println!("{}", line);
    }
}

fn run(data: &GameData, seed: u64, turns: u32) -> Value {
    let started = std::time::Instant::now();
    let france = fid("fac_france");
    let mut state = CampaignState::new_1337(data, france.clone(), seed).expect("state");
    state.interactive_battles = false;
    let orders_log: RefCell<BTreeMap<String, u32>> = RefCell::new(BTreeMap::new());
    let log_order = |o: &Order| {
        let key = match o {
            Order::Recruit { unit_type, .. } => format!("recruit:{}", unit_type.as_str()),
            Order::Build { building, .. } => format!("build:{}", building.as_str()),
            Order::Research { technology } => format!("research:{}", technology.as_str()),
            Order::SetTaxRate { rate } => format!("tax:{rate:?}"),
            Order::SetStance { stance, .. } => format!("stance:{stance:?}"),
            Order::SetCoinage { level } => format!("coinage:{level:?}"),
            other => format!(
                "order:{}",
                format!("{other:?}").split([' ', '{', '(']).next().unwrap_or("?")
            ),
        };
        *orders_log.borrow_mut().entry(key).or_default() += 1;
    };
    let planner = |s: &CampaignState, d: &GameData, f: &FactionId| {
        let orders = ai::plan_turn(s, d, f);
        for o in &orders {
            log_order(o);
        }
        orders
    };
    let mut kinds: BTreeMap<String, u32> = BTreeMap::new();
    let mut per_faction_kind: BTreeMap<String, u32> = BTreeMap::new();
    let mut snaps = serde_json::Map::new();
    snaps.insert("0".into(), snapshot(&state, data));
    let mut comps = serde_json::Map::new();
    comps.insert("0".into(), json!(composition(&state)));
    let mut built: BTreeMap<String, u32> = BTreeMap::new();
    let mut prev_buildings: BTreeSet<(String, String)> = state
        .settlements
        .iter()
        .flat_map(|(sid, s)| {
            s.buildings
                .iter()
                .map(move |b| (sid.as_str().to_owned(), b.as_str().to_owned()))
        })
        .collect();
    let mut war_pairs_prev: BTreeSet<(String, String)> = BTreeSet::new();
    let mut war_start: BTreeMap<(String, String), u32> = BTreeMap::new();
    let mut war_durations: Vec<(String, String, u32, u32)> = Vec::new();
    let mut new_wars = 0u32;
    let mut battles: Vec<Value> = Vec::new();
    let mut captures: BTreeMap<String, u32> = BTreeMap::new();
    let mut owner_changes = 0u32;
    let mut prev_owner: BTreeMap<String, String> = state
        .provinces
        .keys()
        .filter_map(|p| {
            state
                .province_owner(p)
                .map(|o| (p.as_str().to_owned(), o.as_str().to_owned()))
        })
        .collect();
    let mut chronicle_titles: BTreeMap<String, u32> = BTreeMap::new();
    let mut deaths_factions: Vec<(u32, String)> = Vec::new();
    let mut issued = 0u32;
    let mut refused = 0u32;
    let mut per_turn_battles: Vec<u32> = Vec::new();
    let mut revolt_by_faction: BTreeMap<String, u32> = BTreeMap::new();
    let name_to_id: BTreeMap<String, String> = data
        .factions
        .iter()
        .map(|(id, f)| (f.short_or_display_name().to_owned(), id.as_str().to_owned()))
        .collect();

    for t in 0..turns {
        for offer in state.factions[&france].offers.clone() {
            let accept =
                sim_campaign::diplomacy::evaluate(&state, data, &offer.from, &france, &offer.proposal)
                    .accept;
            let _ = state.submit_order(
                data,
                Order::AnswerOffer {
                    offer: offer.id,
                    accept,
                },
            );
        }
        // France's pending decisions: pick the AI option (first).
        for order in ai::plan_turn(&state, data, &france) {
            log_order(&order);
            issued += 1;
            if state.submit_order(data, order).is_err() {
                refused += 1;
            }
        }
        let decisions: Vec<u32> = state
            .chronicle
            .pending_decisions
            .iter()
            .filter(|d| d.faction == france)
            .map(|d| d.id)
            .collect();
        for id in decisions {
            let _ = state.submit_order(
                data,
                Order::ChooseEventOption {
                    decision: id,
                    option: 0,
                },
            );
        }
        let events = state.end_turn_with(data, planner);
        let mut nb = 0;
        for e in &events {
            *kinds.entry(format!("{:?}", e.kind)).or_default() += 1;
            if let Some(f) = &e.faction {
                *per_faction_kind
                    .entry(format!("{:?}|{}", e.kind, f.as_str()))
                    .or_default() += 1;
            }
            match e.kind {
                EventKind::Battle => {
                    nb += 1;
                    // "Bataille de X : A attaque B. Vainqueur : W. Pertes : a contre b."
                    let txt = &e.text_fr;
                    let after = txt.split_once(" : ").map(|x| x.1).unwrap_or("");
                    let (att, rest) = after.split_once(" attaque ").unwrap_or(("", ""));
                    let (def, rest) = rest.split_once(". Vainqueur : ").unwrap_or(("", ""));
                    let (win, rest) = rest.split_once(". Pertes : ").unwrap_or(("", ""));
                    let (la, lb) = rest.trim_end_matches('.').split_once(" contre ").unwrap_or(("0", "0"));
                    let strip = |s: &str| s.split(" (+").next().unwrap_or(s).to_owned();
                    battles.push(json!({
                        "t": t, "att": name_to_id.get(&strip(att)).cloned().unwrap_or(strip(att)),
                        "def": name_to_id.get(&strip(def)).cloned().unwrap_or(strip(def)),
                        "win": name_to_id.get(win).cloned().unwrap_or(win.to_owned()),
                        "la": la.parse::<u32>().unwrap_or(0), "lb": lb.parse::<u32>().unwrap_or(0),
                    }));
                }
                EventKind::Chronicle => {
                    let title = e
                        .text_fr
                        .strip_prefix("Chronique : ")
                        .map(|s| s.trim_end_matches('.').to_owned())
                        .or_else(|| e.text_fr.split_once(" : ").map(|x| x.0.to_owned()))
                        .unwrap_or_else(|| "<autre>".into());
                    *chronicle_titles.entry(title).or_default() += 1;
                }
                EventKind::FactionDestroyed => {
                    deaths_factions.push((t, e.text_fr.clone()));
                }
                EventKind::Revolt => {
                    if let Some(p) = &e.province {
                        let o = state
                            .province_owner(p)
                            .map(|o| o.as_str().to_owned())
                            .unwrap_or_default();
                        *revolt_by_faction.entry(o).or_default() += 1;
                    }
                }
                _ => {}
            }
        }
        per_turn_battles.push(nb);
        // buildings
        let cur: BTreeSet<(String, String)> = state
            .settlements
            .iter()
            .flat_map(|(sid, s)| {
                s.buildings
                    .iter()
                    .map(move |b| (sid.as_str().to_owned(), b.as_str().to_owned()))
            })
            .collect();
        for (_, b) in cur.difference(&prev_buildings) {
            *built.entry(b.clone()).or_default() += 1;
        }
        prev_buildings = cur;
        // wars
        let mut pairs = BTreeSet::new();
        for (id, f) in &state.factions {
            for e in &f.at_war_with {
                if id < e {
                    pairs.insert((id.as_str().to_owned(), e.as_str().to_owned()));
                }
            }
        }
        for p in pairs.difference(&war_pairs_prev) {
            new_wars += 1;
            war_start.insert(p.clone(), t);
        }
        for p in war_pairs_prev.difference(&pairs) {
            let s = war_start.remove(p).unwrap_or(0);
            war_durations.push((p.0.clone(), p.1.clone(), s, t - s));
        }
        war_pairs_prev = pairs;
        // owners
        for p in state.provinces.keys() {
            let o = state
                .province_owner(p)
                .map(|o| o.as_str().to_owned())
                .unwrap_or_default();
            let prev = prev_owner.insert(p.as_str().to_owned(), o.clone());
            if prev.as_deref() != Some(o.as_str()) {
                owner_changes += 1;
                *captures.entry(o).or_default() += 1;
            }
        }
        if (t + 1) % 20 == 0 || t + 1 == turns {
            snaps.insert((t + 1).to_string(), snapshot(&state, data));
            comps.insert((t + 1).to_string(), json!(composition(&state)));
        }
        if state.outcome.is_some() {
            // keep playing: outcome is only the player's
        }
    }
    let techs: BTreeMap<String, Vec<String>> = state
        .factions
        .iter()
        .filter(|(_, f)| f.alive)
        .map(|(id, f)| {
            (
                id.as_str().to_owned(),
                f.technologies.iter().map(|t| t.as_str().to_owned()).collect(),
            )
        })
        .collect();
    let objectives: BTreeMap<String, Value> = ["fac_france", "fac_england", "fac_burgundy"]
        .iter()
        .map(|f| {
            let f = fid(f);
            (
                f.as_str().to_owned(),
                json!({
                    "score": state.campaign_score(data, &f),
                    "done": state.objectives(data, &f).iter().filter(|o| o.done).map(|o| o.id.clone()).collect::<Vec<_>>(),
                    "all": state.objectives(data, &f).iter().map(|o| format!("{}={}", o.id, o.progress)).collect::<Vec<_>>(),
                }),
            )
        })
        .collect();
    let chars_alive = state.characters.values().filter(|c| c.alive).count();
    let traits: BTreeMap<String, u32> = state.characters.values().filter(|c| c.alive).fold(
        BTreeMap::new(),
        |mut m, c| {
            for t in &c.traits {
                *m.entry(t.as_str().to_owned()).or_default() += 1;
            }
            m
        },
    );
    let skills: BTreeMap<String, u32> = state.characters.values().fold(
        BTreeMap::new(),
        |mut m, c| {
            for t in &c.skills_learned {
                *m.entry(t.as_str().to_owned()).or_default() += 1;
            }
            m
        },
    );
    let avg_dev: f64 = state.provinces.values().map(|p| f64::from(p.devastation)).sum::<f64>()
        / state.provinces.len() as f64;
    let avg_unrest: f64 = state.provinces.values().map(|p| f64::from(p.unrest)).sum::<f64>()
        / state.provinces.len() as f64;
    let heresy: usize = state.provinces.values().filter(|p| p.heresy > 0).count();
    json!({
        "seed": seed, "turns": turns, "end": state.date_label(),
        "secs": started.elapsed().as_secs_f64(),
        "kinds": kinds, "per_faction_kind": per_faction_kind,
        "snaps": snaps, "comps": comps, "built": built,
        "orders": *orders_log.borrow(),
        "new_wars": new_wars, "war_durations": war_durations,
        "ongoing_wars": war_start.iter().map(|((a,b),s)| (a.clone(),b.clone(),*s)).collect::<Vec<_>>(),
        "battles": battles, "owner_changes": owner_changes, "captures": captures,
        "chronicle": chronicle_titles, "fired": state.chronicle.fired_events.iter().map(|e| e.as_str().to_owned()).collect::<Vec<_>>(),
        "faction_deaths": deaths_factions, "france_orders": [issued, refused],
        "per_turn_battles": per_turn_battles, "revolts": revolt_by_faction,
        "techs": techs, "objectives": objectives, "outcome": format!("{:?}", state.outcome),
        "chars_alive": chars_alive, "traits": traits, "skills": skills,
        "avg_dev": avg_dev, "avg_unrest": avg_unrest, "heresy_provinces": heresy,
        "schism": state.schism,
    })
}
